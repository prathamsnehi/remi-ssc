import CoreVideo
import CoreImage

// ⚠️ PERFORMANCE TIP: Create this ONCE and reuse it.
// Creating a CIContext is expensive (heavy on CPU/GPU).
let sharedContext = CIContext()

func renderToPixelBuffer(from ciImage: CIImage, targetSize: CGSize) -> CVPixelBuffer? {

    // 1. Attributes for the new Pixel Buffer
    // We explicitly demand BGRA to match your SFace model's expectations.
    let attrs = [
        kCVPixelBufferCGImageCompatibilityKey: kCFBooleanTrue,
        kCVPixelBufferCGBitmapContextCompatibilityKey: kCFBooleanTrue,
        kCVPixelBufferWidthKey: Int(targetSize.width),
        kCVPixelBufferHeightKey: Int(targetSize.height),
        kCVPixelBufferPixelFormatTypeKey: kCVPixelFormatType_32BGRA // <--- CRITICAL for SFace
    ] as CFDictionary

    // 2. Allocate the Buffer
    var pixelBuffer: CVPixelBuffer?
    let status = CVPixelBufferCreate(kCFAllocatorDefault,
                                     Int(targetSize.width),
                                     Int(targetSize.height),
                                     kCVPixelFormatType_32BGRA,
                                     attrs,
                                     &pixelBuffer)

    guard status == kCVReturnSuccess, let buffer = pixelBuffer else {
        print("❌ Failed to allocate pixel buffer: \(status)")
        return nil
    }

    // 3. Render the CIImage into the Buffer
    // This effectively "draws" your processed image (cropped/scaled) into the memory.
    sharedContext.render(ciImage, to: buffer)

    return buffer
}
